// Author: @patriciogv // Title: 4 cells voronoi #ifdef GL_ES precision mediump float; #endif uniform vec2 u_resolution; uniform vec2 u_mouse; uniform float u_time; float random (in vec2 st) { return fract(sin(dot(st.xy, vec2(12.9898, 78.233))) * 43758.5453123); } float noise (in vec2 st) { vec2 i = floor(st); vec2 f = fract(st); float a = random(i); float b = random(i + vec2(1.0, 0.0)); float c = random(i + vec2(0.0, 1.0)); float d = random(i + vec2(1.0, 1.0)); vec2 u = f * f * (3.0 - 2.0 * f); return mix(a, b, u.x) + (c - a) * u.y * (1.0 - u.x) + (d - b) * u.x * u.y; } #define OCTAVES 6 float fbm (in vec2 st) { float value = 0.0; float amplitude = .8; for (int i = 0; i < OCTAVES; i++) { value += amplitude * noise(st); st *= 2.; amplitude *= .3; } return value; } void main() { vec2 st = gl_FragCoord.xy / u_resolution.xy; st.x *= u_resolution.x / u_resolution.y; vec3 color = vec3(0.255, 0.255, 0.255); vec2 point[9]; point[0] = vec2(0.83, 0.75); point[1] = vec2(0.60, 0.07); point[2] = vec2(0.920, 0.930); point[3] = vec2(0.31, 0.26); point[4] = u_mouse / u_resolution; point[5] = vec2(0.220, 0.400); point[6] = vec2(0.410, 0.660); point[7] = vec2(0.720, 0.500); point[8] = vec2(0.920, 0.130); float m_dist = 1.0; float m_dist2 = 1.0; vec2 m_point; for (int i = 0; i < 9; i++) { float dist = distance(st, point[i]); if (dist < m_dist) { m_dist2 = m_dist; m_dist = dist; m_point = point[i]; } else if (dist < m_dist2) { m_dist2 = dist; } } color += m_dist * 2.4; color.rg = m_point; // --- Surface texture --- // fbm() gives us a soft, organic mottling instead of one flat tint per // cell - like faint thickness variations on real soap film. Scaled up // (st*25) so the detail is fine grain rather than one big blob, and // centered around 0 (subtract 0.5) so it can lighten OR darken instead // of just brightening everything. float grain = fbm(st * 15.0); color += (grain - 0.2) * 0.2; // --- Border controls --- float edge = m_dist2 - m_dist; float outline_alpha = (0.80 - smoothstep(0.002, 0.02, edge)) * 0.704; vec3 tintA = vec3(0.85, 0.95, 1.0); vec3 tintB = vec3(0.95, 0.85, 1.0); vec3 tintC = vec3(0.85, 1.0, 0.9); vec3 tintD = vec3(1.000, 0.717, 0.002); vec3 haloColor = mix(tintA, tintC, m_point.x); haloColor = mix(haloColor, tintD, m_point.y); haloColor = mix(haloColor, tintA, m_point.y); color = mix(color, haloColor, outline_alpha); color += (0.99 - smoothstep(0.3, 0., edge)) * 3.2; gl_FragColor = vec4(clamp(color, .0, 1.0), 0.90); }
